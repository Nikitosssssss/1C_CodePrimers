&НаСервере
Процедура СоздатьУстановкуЦенНаСервере()		//для ценообразования 2.5
	
	НовыйДокумент = Документы.УстановкаЦенНоменклатуры.СоздатьДокумент();
	НовыйДокумент.Дата = ТекущаяДата(); 
	НовыйДокумент.Согласован = Истина;
	НовыйДокумент.Статус = Перечисления.СтатусыУстановокЦенНоменклатуры.Согласован;
	НовыйДокумент.Ответственный = Пользователи.ТекущийПользователь();
	НовыйДокумент.Комментарий = "Создано программно из внешней обработки " + ТекущаяДата();
	Для каждого ВидЦены из Объект.ВидыЦен цикл		
		НоваяСтрока = НовыйДокумент.ВидыЦен.Добавить();
		НоваяСтрока.ВидЦены = ВидЦены.Значение; 	
		
		// Пример заполнения строки товара
		НоваяСтрока = НовыйДокумент.Товары2_5.Добавить();
		НоваяСтрока.Номенклатура = Объект.Номенклатура;
		НоваяСтрока.ХарактеристикаЦО = Объект.Характеристика.ХарактеристикаНоменклатурыДляЦенообразования;
		НоваяСтрока.ВидЦены = ВидЦены.Значение; 
		НоваяСтрока.Цена = 1500; // Ваша цена
		
	КонецЦикла;
	НовыйДокумент.Записать();
	Сообщение = Новый СообщениеПользователю();
	Сообщение.КлючДанных = НовыйДокумент.Ссылка;
	Сообщение.Текст = "Документ записан!";
	Сообщение.Сообщить();
	
	
КонецПроцедуры
