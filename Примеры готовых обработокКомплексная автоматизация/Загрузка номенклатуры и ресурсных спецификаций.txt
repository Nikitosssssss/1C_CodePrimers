&НаКлиенте
Процедура СоздатьНоменклатуру(Команда)
	
	СозданнаяНоменклатура.Очистить(); 
	НоменклатураСРаботами.Очистить();
	СозданныеСпецификации.Очистить();
	Если Не ЗначениеЗаполнено(РежимЗагрузки) тогда
		
		Сообщение = Новый СообщениеПользователю();
		Сообщение.Текст = "Необходимо заполнить режим загрузки";
		Сообщение.Поле = "РежимЗагрузки";
		Сообщение.Сообщить();
		Возврат; 
		
	КонецЕсли;

	Если РежимЗагрузки = "Металлы" тогда
		Если Не ЗначениеЗаполнено(ВидНоменклатуры) тогда			
			Сообщение = Новый СообщениеПользователю();
			Сообщение.Текст = "Необходимо заполнить вид номенклатуры";
			Сообщение.Поле = "ВидНоменклатуры";
			Сообщение.Сообщить();
			Возврат; 		
		КонецЕсли;
		СоздатьНоменклатуруНаСервере();
	ИначеЕсли РежимЗагрузки = "Метизы" тогда
		Если Не ЗначениеЗаполнено(ВидНоменклатуры) тогда			
			Сообщение = Новый СообщениеПользователю();
			Сообщение.Текст = "Необходимо заполнить вид номенклатуры";
			Сообщение.Поле = "ВидНоменклатуры";
			Сообщение.Сообщить();
			Возврат; 		
		КонецЕсли;
		СоздатьМетизы();
	ИначеЕсли РежимЗагрузки = "Подшипники" тогда
		Если Не ЗначениеЗаполнено(ВидНоменклатуры) тогда			
			Сообщение = Новый СообщениеПользователю();
			Сообщение.Текст = "Необходимо заполнить вид номенклатуры";
			Сообщение.Поле = "ВидНоменклатуры";
			Сообщение.Сообщить();

		Если Не ЗначениеЗаполнено(Подразделение) тогда			
			Сообщение = Новый СообщениеПользователю();
			Сообщение.Текст = "Необходимо заполнить подразделение";
			Сообщение.Поле = "Подразделение";
			Сообщение.Сообщить();
			Возврат; 		
		КонецЕсли;
		СоздатьНоменклатуруДляСпецификации()

		Если Не ЗначениеЗаполнено(ВидНоменклатуры) тогда			
			Сообщение = Новый СообщениеПользователю();

			Сообщение.Поле = "ВидНоменклатуры";

			Содержимое = Ячейка.Текст;
			Если Содержимое <> ""  тогда	
				Если МассивЗаголовков[НомерКолонки-1] <> "Материал" тогда 
					Колонка = ЗагружаемаяНоменклатура.Колонки.Найти(СтрЗаменить(МассивЗаголовков[НомерКолонки-1]," ", ""));
					Если Колонка = Неопределено тогда					
						ЗагружаемаяНоменклатура.Колонки.Добавить(МассивЗаголовков[НомерКолонки-1], Новый ОписаниеТипов("Строка"));

				ЗначениеРеквизита = ПолучитьДополнительноеЗначение(Реквизит, ТекущаяСтрока[Реквизит.Свойство.ИдентификаторДляФормул]);

			НоваяСпецификация.ОсновноеИзделиеВидНоменклатуры = НоменклатураДляОтбора.ВидНоменклатуры;
			НоваяСпецификация.ВыпускПроизвольнымиПорциями = Истина;
			СтрокаТЧВыходныеИзделия = НоваяСпецификация.ВыходныеИзделия.Добавить();
			СтрокаТЧВыходныеИзделия.Номенклатура = НоменклатураДляОтбора;
			СтрокаТЧВыходныеИзделия.КоличествоУпаковок = 1;
			СтрокаТЧВыходныеИзделия.ВидНоменклатуры =НоменклатураДляОтбора.ВидНоменклатуры;
			СтрокаТЧВыходныеИзделия.ОбработатьПоСпецификации = Ложь;
			СтрокаТЧВыходныеИзделия.СпособАвтовыбораНоменклатуры =СпособАвтовыбораНоменклатуры ;
			СтрокаТЧВыходныеИзделия.СпособАвтовыбораХарактеристики = СпособАвтовыбораХарактеристики; 
			
			ПараметрыОтбора = Новый Структура("Номенклатура", НоменклатураДляОтбора);
			ТехРаботы = ТаблицаСНоменклатурой.НайтиСтроки(ПараметрыОтбора);
			
			//заполняем материалы 
			//Если это последняя строка, в нее все закидываем
			Если СтрокаТЗ = СозданнаяНоменклатура[СозданнаяНоменклатура.Количество() -1] тогда				
				
				Для каждого Исходник из МассивПростыхПФ цикл
					
					Материал = НоваяСпецификация.МатериалыИУслуги.Добавить();
					Материал.Номенклатура = Исходник.Номенклатура;
					Материал.КоличествоУпаковок = Исходник.Количество;	
					Материал.СтатьяКалькуляции = СтатьяПолуфабрикат;	
					Материал.СпособПолученияМатериала = СпособПолучения;
					Материал.Упаковка = СтрокаТЗ.ЕдиницаИзмерения;
				КонецЦикла;
				
			Иначе
				НоменклатураИКоличество = Новый Структура;
				НоменклатураИКоличество.Вставить("Номенклатура",НоменклатураДляОтбора);    
				НоменклатураИКоличество.Вставить("Количество",СтрокаТЗ.КоличествоИзделий); 
				МассивПростыхПФ.Добавить(НоменклатураИКоличество);
			КонецЕсли;
			Если НоменклатураДляОтбора.Видноменклатуры = ПФПростой тогда
				
				Исходник = Справочники.Номенклатура.ПустаяСсылка();
				ДопРеквизиты = НоменклатураДляОтбора.ДополнительныеРеквизиты;
				Для каждого ДопРеквизит из ДопРеквизиты цикл
					//если это исходник
					Если ДопРеквизит.Свойство = ПланыВидовХарактеристик.ДополнительныеРеквизитыИСведения.ПолучитьСсылку(Новый УникальныйИдентификатор("4e91184e-ae5a-11f0-921a-00155d46fb81")) тогда	
						Исходник = ДопРеквизит.Значение;		
					КонецЕсли;
				КонецЦикла;	
				
				Если ЗначениеЗаполнено(Исходник) тогда
					
					Материал = НоваяСпецификация.МатериалыИУслуги.Добавить();
					Материал.Номенклатура = Исходник;
					Материал.КоличествоУпаковок = СтрокаТЗ.ВесИсходника;	
					Материал.СтатьяКалькуляции = СтатьяМПЗ;  
					Материал.СпособПолученияМатериала = СпособПолучения;
					Материал.Упаковка = Исходник.ЕдиницаИзмерения;
				КонецЕсли;
				
			КонецЕсли; 
			
			Для каждого Работа из ТехРаботы цикл   //заполняем работы
				НаименованиеЭтапа = НаименованиеЭтапа + Работа.Работа + ","; 
				//Сказали не заполнять работы
				//СтрокаРабота = НоваяСпецификация.Трудозатраты.Добавить();
				//СтрокаРабота.ВидРабот = Работа.Работа;
				//СтрокаРабота.Количество = 1;
				//СтрокаРабота.СтатьяКалькуляции = СтатьяТрудозатраты;		
			КонецЦикла;
			
			
			Попытка
				НоваяСпецификация.Записать(); 		
				//добавляем этап
				НаименованиеЭтапа = Лев(НаименованиеЭтапа, СтрДлина(НаименованиеЭтапа)-1);
				НовыйЭтап = Справочники.ЭтапыПроизводства.СоздатьЭлемент();
				НовыйЭтап.Владелец = НоваяСпецификация.Ссылка;
				НовыйЭтап.Подразделение = Подразделение;
				НовыйЭтап.Наименование = НаименованиеЭтапа; 
				НовыйЭтап.ЕдиницаИзмеренияПредварительногоБуфера = ЕдиницаДень;
				НовыйЭтап.ЕдиницаИзмеренияЗавершающегоБуфера = ЕдиницаДень;
				Попытка
					НовыйЭтап.Записать();
				Исключение
					Сообщить(КраткоеПредставлениеОшибки(ИнформацияОбОшибке())); 	
				КонецПопытки;
				
				НоваяСпецификация.ОсновноеИзделиеЭтап =  НовыйЭтап.Ссылка; 
				НоваяСпецификация.Описание = НаименованиеЭтапа;
				НоваяСпецификация.Статус = Перечисления.СтатусыСпецификаций.Действует;
				НоваяСпецификация.НачалоДействия = ТекущаяДата();
				НоваяСпецификация.Записать();
				НоваяСтрока = СозданныеСпецификации.Добавить();
				НоваяСтрока.Спецификация= НоваяСпецификация.Ссылка; 
				Сообщить("Спецификация создана: " + НоваяСпецификация.Наименование);
				
			Исключение
				Сообщить("Не удолось записать спецификацию " + НоменклатураДляОтбора.Наименование + " по причине: " + КраткоеПредставлениеОшибки(ИнформацияОбОшибке())); 	
			КонецПопытки;
		КонецЕсли;
	КонецЦикла;
		
КонецПроцедуры
