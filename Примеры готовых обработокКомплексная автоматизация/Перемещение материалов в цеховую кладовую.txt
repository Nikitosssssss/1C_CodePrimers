Процедура СоздатьПеремещение(ЦеховаяКладовая, СписокНоменклатуры)	
	Если СписокНоменклатуры.Количество() = 0 тогда
		Возврат;
	КонецЕсли;
	НовыйДокумент = Документы.ДвижениеПродукцииИМатериалов.СоздатьДокумент();   
	НовыйДокумент.Дата = ДатаПолученияОстатка; 
	НовыйДокумент.ВариантПриемкиТоваров = Перечисления.ВариантыПриемкиТоваров.РазделенаТолькоПоНакладным;
	НовыйДокумент.Отправитель = Склад; 
	НовыйДокумент.Получатель = ЦеховаяКладовая;
	НовыйДокумент.ХозяйственнаяОперация = Перечисления.ХозяйственныеОперации.ПередачаМатериаловВКладовую;            
	НовыйДокумент.Организация = Организация;
	НовыйДокумент.НаправлениеДеятельности = НаправлениеДеятельности;
	НовыйДокумент.Статус = Перечисления.СтатусыДвиженияПродукцииИМатериалов.Принято;
	Автор = ПараметрыСеанса.ТекущийПользователь;
	НовыйДокумент.Ответственный = Автор;
	НовыйДокумент.Автор = Автор;
	Для каждого СтрокаТЧ из СписокНоменклатуры Цикл 
		
		СтрокаНоменклатуры = НовыйДокумент.Товары.Добавить();
		  
		СтрокаНоменклатуры.Номенклатура = СтрокаТЧ.Номенклатура;
		СтрокаНоменклатуры.Упаковка = СтрокаТЧ.Упаковка; 
		СтрокаНоменклатуры.Характеристика =  СтрокаТЧ.Характеристика; 
			 
		СтрокаНоменклатуры.КоличествоУпаковок = СтрокаТЧ.Количество;
		СтрокаНоменклатуры.Количество = СтрокаТЧ.Количество;
	КонецЦикла;
	Попытка 
		НовыйДокумент.Записать(РежимЗаписиДокумента.Проведение);
		Сообщить("Документ " +НовыйДокумент.Ссылка + " проведен.");
		НоваяСтрока = ТаблицаДокументов.Добавить(); 
		НоваяСтрока.Документ = НовыйДокумент.Ссылка;  
		НоваяСтрока.Склад = ЦеховаяКладовая; 
	Исключение
		Сообщить(КраткоеПредставлениеОшибки(ИнформацияОбОшибке()));
	КонецПопытки;
