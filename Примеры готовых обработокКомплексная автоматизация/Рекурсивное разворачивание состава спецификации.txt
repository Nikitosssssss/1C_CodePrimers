&НаКлиенте
Процедура ЗаполнитьМатериалы(Команда)
	
	Если Не ЗначениеЗаполнено(Спецификация) тогда
		
		Сообщение = Новый СообщениеПользователю();
		Сообщение.ПутьКДанным = "Спецификация";
		Сообщение.Текст = "Необходимо заполнить спецификацию!";
		Сообщение.Сообщить();	
		Возврат;
		
	КонецЕсли;
	ДеревоСпецификаций.ПолучитьЭлементы().Очистить();
	ЗаполнитьМатериалыНаСервере();
	
КонецПроцедуры

&НаСервере
Процедура ЗаполнитьМатериалыНаСервере()
	
	ДеревоОбъект = РеквизитФормыВЗначение("ДеревоСпецификаций");
	
	Для каждого РС из Спецификация цикл
		
		Запрос = Новый Запрос;
		Запрос.Текст = 
		"ВЫБРАТЬ ПЕРВЫЕ 1
		|	РесурсныеСпецификации.Ссылка КАК Спецификация,
		|	ТоварыНаСкладахОстатки.ВНаличииОстаток КАК Остаток,
		|	РесурсныеСпецификации.ОсновноеИзделиеНоменклатура КАК Номенклатура
		|ИЗ
		|	Справочник.РесурсныеСпецификации КАК РесурсныеСпецификации
		|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрНакопления.ТоварыНаСкладах.Остатки КАК ТоварыНаСкладахОстатки
		|		ПО РесурсныеСпецификации.ОсновноеИзделиеНоменклатура = ТоварыНаСкладахОстатки.Номенклатура
		|			И РесурсныеСпецификации.ОсновноеИзделиеХарактеристика = ТоварыНаСкладахОстатки.Характеристика
		|ГДЕ
		|	РесурсныеСпецификации.Ссылка = &Спецификация";
		
		Запрос.УстановитьПараметр("Спецификация", РС.Значение);
		
		РезультатЗапроса = Запрос.Выполнить();
		
		ВыборкаДетальныеЗаписи = РезультатЗапроса.Выбрать();
		
		Пока ВыборкаДетальныеЗаписи.Следующий() Цикл
			НоваяСтрока = ДеревоОбъект.Строки.Добавить();
			НоваяСтрока.Номенклатура = ВыборкаДетальныеЗаписи.Номенклатура;
			НоваяСтрока.Количество = 1; 
			НоваяСтрока.Спецификация = ВыборкаДетальныеЗаписи.Спецификация;
			НоваяСтрока.Остаток = ВыборкаДетальныеЗаписи.Остаток;
			
			ОбойтиСпецификацииРекурсивно(ВыборкаДетальныеЗаписи.Спецификация, 1, НоваяСтрока);
		КонецЦикла;
			
	КонецЦикла;
	
	ЗначениеВРеквизитФормы(ДеревоОбъект,"ДеревоСпецификаций");
		
КонецПроцедуры

&НаСервере
Процедура ОбойтиСпецификацииРекурсивно(СпецификацияРодителя, КоличествоРодителя, СтрокаДЗ) 

	Запрос = Новый Запрос;
	Запрос.Текст = 
		"ВЫБРАТЬ
		|	РесурсныеСпецификацииМатериалыИУслуги.Ссылка КАК Спецификация,
		|	РесурсныеСпецификацииМатериалыИУслуги.Номенклатура КАК Номенклатура,
		|	РесурсныеСпецификацииМатериалыИУслуги.Характеристика КАК Характеристика,
		|	РесурсныеСпецификацииМатериалыИУслуги.КоличествоУпаковок*&КоличествоРодителя КАК Количество,
		|	РесурсныеСпецификации.Ссылка КАК СпецификацияПФ,
		|   ТоварыНаСкладахОстатки.ВНаличииОстаток КАК Остаток
		|ИЗ
		|	Справочник.РесурсныеСпецификации.МатериалыИУслуги КАК РесурсныеСпецификацииМатериалыИУслуги
		|		ЛЕВОЕ СОЕДИНЕНИЕ Справочник.РесурсныеСпецификации КАК РесурсныеСпецификации
		|		ПО РесурсныеСпецификацииМатериалыИУслуги.Номенклатура = РесурсныеСпецификации.ОсновноеИзделиеНоменклатура
		|			И РесурсныеСпецификацииМатериалыИУслуги.Характеристика = РесурсныеСпецификации.ОсновноеИзделиеХарактеристика
		|		ЛЕВОЕ СОЕДИНЕНИЕ РегистрНакопления.ТоварыНаСкладах.Остатки КАК ТоварыНаСкладахОстатки
		|		ПО (РесурсныеСпецификацииМатериалыИУслуги.Номенклатура = ТоварыНаСкладахОстатки.Номенклатура
		|			И РесурсныеСпецификацииМатериалыИУслуги.Характеристика = ТоварыНаСкладахОстатки.Характеристика)
		|ГДЕ
		|	РесурсныеСпецификацииМатериалыИУслуги.Ссылка = &Спецификация
		|
		|УПОРЯДОЧИТЬ ПО
		|	СпецификацияПФ";
	
	Запрос.УстановитьПараметр("Спецификация", СпецификацияРодителя);  
	Запрос.УстановитьПараметр("КоличествоРодителя", КоличествоРодителя);
	
	РезультатЗапроса = Запрос.Выполнить();
	Выборка = РезультатЗапроса.Выбрать();
	
	Пока Выборка.Следующий() Цикл		
		
		НоваяСтрока = СтрокаДЗ.Строки.Добавить();
		НоваяСтрока.Номенклатура = Выборка.Номенклатура;
		НоваяСтрока.Количество = Выборка.Количество; 
		НоваяСтрока.Спецификация = Выборка.Спецификация;	
		НоваяСтрока.Остаток = Выборка.Остаток;
		Если Выборка.СпецификацияПФ <> NULL тогда
			
			ОбойтиСпецификацииРекурсивно(Выборка.СпецификацияПФ, Выборка.Количество, НоваяСтрока);	
			
		КонецЕсли;
		
	КонецЦикла;
	
КонецПроцедуры
