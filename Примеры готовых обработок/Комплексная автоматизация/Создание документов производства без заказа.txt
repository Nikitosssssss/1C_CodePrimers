Функция ОбработатьЧисло(ЧислоСтрокой)
	
	//в строку попадает лишний символ '
	СтрокаРезультат = "";
	Если СтрДлина(ЧислоСтрокой) = 0 тогда
		Возврат 0;
	КонецЕсли;
	Для НомерСимвола = 1 по СтрДлина(ЧислоСтрокой) цикл
		Символ = Сред(ЧислоСтрокой,НомерСимвола, 1);
		Если СтрНайти("1234567890,.",Символ) <> 0 тогда		
			СтрокаРезультат = СтрокаРезультат + Символ; 	
		КонецЕсли;	
	КонецЦикла;
	Попытка
		Результат = Число(СтрокаРезультат);	
	Исключение
		 Результат = 0;
	КонецПопытки;
 	Возврат Результат;

Функция УбратьлишниеСимволы(ВходящаяСтрока)

Функция ПолучитьВремяВСекундах(ВремяПроизводства, Количество) 
	
	Результат = 0;
	МассивВремени = СтрРазделить(ВремяПроизводства, ":");
	Попытка
		Часы = МассивВремени[0];
		Минуты = МассивВремени[1];
		Секунды = МассивВремени[2];
		Результат = (Часы*60*60 + Минуты * 60 + Секунды) * Количество;
	Исключение
		Сообщить("Не удалось получить время из ячейки!");	
	КонецПопытки;
	
	Возврат Результат;

Функция ПолучитьВес(ВесДетали, Количество) 
	
	ВесДетали= СокрЛП(ВесДетали);
	ВесДетали = СтрЗаменить(ВесДетали," kg", "");
	ВесДеталиЧисло = ОбработатьЧисло(ВесДетали);
	ВесДетали = ВесДеталиЧисло*Количество;
	Возврат ВесДетали;
